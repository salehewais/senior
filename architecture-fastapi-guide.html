Dependency Rule

Layered Architecture

تقسيم رأسي شائع: Presentation → Business → Data. كل طبقة تتكلم مع اللي تحتها مباشرة. أبسط للفهم وأسرع للبدء في أغلب مشاريع CRUD المتوسطة.

Clean Architecture

قلب مستقل (Domain) محاط بـ Use Cases، والـ Infrastructure/UI على الأطراف. الاعتماديات تتجه للداخل فقط — الـ Domain لا يعرف FastAPI ولا SQLAlchemy.

قاعدة الاعتمادية في Clean: الكود الداخلي لا يعتمد على الخارجي. الـ Domain لا يستورد من Infrastructure؛ والعكس مسموح عبر Interfaces (Ports).

قاعدة الاعتمادية في Layered: من فوق لتحت — الـ API يعتمد على Service، والـ Service على Repository، والـ Repository على الـ Database/ORM.

اختر Layered لو الفريق صغير والمشروع واضح الحدود. انتقل لـ Clean لما تتعدد مصادر البيانات، أو تحتاج اختبارات Use Case بدون DB، أو يتغير الـ framework كثيرًا.

repositories/todo_repository.pypython
from typing import List, Optional
from uuid import UUID
from sqlalchemy import select, delete
from sqlalchemy.ext.asyncio import AsyncSession
from models.todo import Todo

class TodoRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def create(self, title: str, description: str | None = None) -> Todo:
        todo = Todo(title=title, description=description)
        self.session.add(todo)
        await self.session.commit()
        await self.session.refresh(todo)
        return todo

    async def get_by_id(self, todo_id: UUID) -> Optional[Todo]:
        result = await self.session.execute(
            select(Todo).where(Todo.id == todo_id)
        )
        return result.scalar_one_or_none()

    async def get_all(self) -> List[Todo]:
        result = await self.session.execute(select(Todo))
        return list(result.scalars().all())

    async def update(
        self,
        todo_id: UUID,
        title: str | None = None,
        description: str | None = None,
        is_completed: bool | None = None,
    ) -> Optional[Todo]:
        todo = await self.get_by_id(todo_id)
        if not todo:
            return None

        if title is not None:
            todo.title = title
        if description is not None:
            todo.description = description
        if is_completed is not None:
            todo.is_completed = is_completed

        await self.session.commit()
        await self.session.refresh(todo)
        return todo

    async def delete(self, todo_id: UUID) -> bool:
        result = await self.session.execute(
            delete(Todo).where(Todo.id == todo_id)
        )
        await self.session.commit()
        return result.rowcount > 0

المنطق التجاري يعيش هنا؛ الـ API يستدعي الـ Service فقط.

services/todo_service.pypython
from typing import List, Optional
from uuid import UUID
from sqlalchemy.ext.asyncio import AsyncSession
from repositories.todo_repository import TodoRepository
from models.todo import Todo

class TodoService:
    def __init__(self, session: AsyncSession):
        self.repository = TodoRepository(session)

    async def create_todo(self, title: str, description: str | None = None) -> Todo:
        # هنا تقدر تضيف أي Business Logic
        # مثلاً: validation إضافي، logging، إرسال إشعار... إلخ
        if not title.strip():
            raise ValueError("Title cannot be empty")

        return await self.repository.create(title=title, description=description)

    async def get_todo(self, todo_id: UUID) -> Optional[Todo]:
        return await self.repository.get_by_id(todo_id)

    async def list_todos(self) -> List[Todo]:
        return await self.repository.get_all()

    async def update_todo(
        self,
        todo_id: UUID,
        title: str | None = None,
        description: str | None = None,
        is_completed: bool | None = None,
    ) -> Optional[Todo]:
        return await self.repository.update(
            todo_id=todo_id,
            title=title,
            description=description,
            is_completed=is_completed,
        )

    async def delete_todo(self, todo_id: UUID) -> bool:
        return await self.repository.delete(todo_id)
api/todos.py
api/todos.pypython
from fastapi import APIRouter, Depends, HTTPException, status
from uuid import UUID
from sqlalchemy.ext.asyncio import AsyncSession

from database import get_db
from services.todo_service import TodoService
from schemas.todo import TodoCreate, TodoUpdate, TodoResponse

router = APIRouter(prefix="/todos", tags=["Todos"])

def get_service(session: AsyncSession = Depends(get_db)) -> TodoService:
    return TodoService(session)

@router.post("/", response_model=TodoResponse, status_code=status.HTTP_201_CREATED)
async def create_todo(
    data: TodoCreate,
    service: TodoService = Depends(get_service),
):
    try:
        todo = await service.create_todo(title=data.title, description=data.description)
        return todo
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))

@router.get("/", response_model=list[TodoResponse])
async def list_todos(service: TodoService = Depends(get_service)):
    return await service.list_todos()

@router.get("/{todo_id}", response_model=TodoResponse)
async def get_todo(
    todo_id: UUID,
    service: TodoService = Depends(get_service),
):
    todo = await service.get_todo(todo_id)
    if not todo:
        raise HTTPException(status_code=404, detail="Todo not found")
    return todo

@router.put("/{todo_id}", response_model=TodoResponse)
async def update_todo(
    todo_id: UUID,
    data: TodoUpdate,
    service: TodoService = Depends(get_service),
):
    todo = await service.update_todo(
        todo_id=todo_id,
        title=data.title,
        description=data.description,
        is_completed=data.is_completed,
    )
    if not todo:
        raise HTTPException(status_code=404, detail="Todo not found")
    return todo

@router.delete("/{todo_id}", status_code=status.HTTP_204_NO_CONTENT)
async def delete_todo(
    todo_id: UUID,
    service: TodoService = Depends(get_service),
):
    success = await service.delete_todo(todo_id)
    if not success:
        raise HTTPException(status_code=404, detail="Todo not found")

        pass

    @abstractmethod
    async def list_all(self) -> List[Todo]:
        pass

infrastructure/repositories/todo_repository_impl.pypython
from typing import List, Optional
from uuid import UUID
from sqlalchemy import select, delete
from sqlalchemy.ext.asyncio import AsyncSession
from domain.entities.todo import Todo
from domain.repositories.todo_repository import TodoRepository
from infrastructure.database.models import TodoModel

class TodoRepositoryImpl(TodoRepository):
    def __init__(self, session: AsyncSession):
        self.session = session

    async def create(self, todo: Todo) -> Todo:
        db_todo = TodoModel(
            id=todo.id,
            title=todo.title,
            description=todo.description,
            is_completed=todo.is_completed,
            created_at=todo.created_at,
            updated_at=todo.updated_at,
        )
        self.session.add(db_todo)
        await self.session.commit()
        await self.session.refresh(db_todo)
        return self._to_entity(db_todo)

    async def get_by_id(self, todo_id: UUID) -> Optional[Todo]:
        result = await self.session.execute(
            select(TodoModel).where(TodoModel.id == todo_id)
        )
        db_todo = result.scalar_one_or_none()
        return self._to_entity(db_todo) if db_todo else None

    async def list_all(self) -> List[Todo]:
        result = await self.session.execute(select(TodoModel))
        return [self._to_entity(item) for item in result.scalars().all()]

    async def update(self, todo: Todo) -> Todo:
        result = await self.session.execute(
            select(TodoModel).where(TodoModel.id == todo.id)
        )
        db_todo = result.scalar_one()
        db_todo.title = todo.title
        db_todo.description = todo.description
        db_todo.is_completed = todo.is_completed
        db_todo.updated_at = todo.updated_at
        await self.session.commit()
        await self.session.refresh(db_todo)
        return self._to_entity(db_todo)

    async def delete(self, todo_id: UUID) -> bool:
        result = await self.session.execute(
            delete(TodoModel).where(TodoModel.id == todo_id)
        )
        await self.session.commit()
        return result.rowcount > 0

    def _to_entity(self, model: TodoModel) -> Todo:
        return Todo(
            id=model.id,
            title=model.title,
            description=model.description,
            is_completed=model.is_completed,
            created_at=model.created_at,
            updated_at=model.updated_at,
        )

@router.get("/{todo_id}", response_model=TodoResponse)

    todo_id: UUID,
    repo: TodoRepositoryImpl = Depends(get_repository),
):
    use_case = GetTodoUseCase(repo)
    todo = await use_case.execute(todo_id)

1) Exception Handling مركزي

لا ترفع HTTPException من الـ Domain/Use Cases. عرّف أخطاء مجال، وترجمها لـ HTTP في طبقة العرض عبر exception handlers.

exception handling sketchpython
# domain/errors.py
class DomainError(Exception):
    """Base domain error."""

class NotFoundError(DomainError):
    def __init__(self, resource: str, id: str):
        self.resource = resource
        self.id = id
        super().__init__(f"{resource} not found: {id}")

class ValidationError(DomainError):
    pass


# presentation/exception_handlers.py
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
from domain.errors import NotFoundError, ValidationError, DomainError

def register_exception_handlers(app: FastAPI) -> None:
    @app.exception_handler(NotFoundError)
    async def not_found_handler(_: Request, exc: NotFoundError):
        return JSONResponse(status_code=404, content={"detail": str(exc)})

    @app.exception_handler(ValidationError)
    async def validation_handler(_: Request, exc: ValidationError):
        return JSONResponse(status_code=400, content={"detail": str(exc)})

    @app.exception_handler(DomainError)
    async def domain_handler(_: Request, exc: DomainError):
        return JSONResponse(status_code=422, content={"detail": str(exc)})

2) Dependency Injection

FastAPI Depends يكفي لمعظم المشاريع: يحقن الـ session والـ repository والـ use case في الـ route. مكتبة dependency-injector اختيارية لما يصير الـ wiring معقّدًا (عدة adapters، lifespan، اختبارات كثيرة).

3) قائمة ترحيل Layered → Clean

  • استخرج Domain Entity منفصل عن SQLAlchemy Model (مثلاً Todo vs TodoModel).
  • انقل عقد التخزين إلى Interface في domain/repositories واترك التنفيذ في Infrastructure.
  • قسّم TodoService إلى Use Cases صغيرة (Create / Update / …).
  • اجعل الـ API يبني Use Case فقط — بدون منطق أعمال داخل الـ route.
  • أضف mapper (_to_entity) في الـ Repository Impl.
  • انقل أخطاء المجال خارج HTTPException وسجّل handlers مركزية.
  • اكتب اختبارات Use Case بـ fake repository قبل لمس قاعدة البيانات.
